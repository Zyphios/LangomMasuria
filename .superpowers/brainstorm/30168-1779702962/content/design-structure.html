docker-compose.yml — usługi

frontend
image: node:20-alpine
port: 3000:3000
build: ./frontend
depends_on: backend
backend
image: node:20-alpine
port: 4000:4000
build: ./backend
depends_on: db
env_file: .env
db
image: postgres:16
port: 5432:5432
volume: postgres_data
env: POSTGRES_DB/USER/PASSWORD
.env — SMTP_*, DATABASE_URL, JWT_SECRET, OWNER_EMAIL, PAYU_* (zaślepka)
